cmake_minimum_required(VERSION 3.20)
project(BackbreakerRecomp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

if (NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Recompiled code requires Clang (configure with -DCMAKE_CXX_COMPILER=clang++)")
endif()

set(XENON ${CMAKE_SOURCE_DIR}/third_party/XenonRecomp)
set(COMMON_INCLUDES ${CMAKE_SOURCE_DIR}/ppc ${XENON}/XenonUtils ${XENON}/thirdparty/simde)
set(COMMON_FLAGS -march=x86-64-v3 -fno-strict-aliasing)

# Recompiled game code (generated by XenonRecomp into ppc/)
file(GLOB PPC_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/ppc/*.cpp)
add_library(backbreaker_ppc STATIC ${PPC_SOURCES})
target_include_directories(backbreaker_ppc PUBLIC ${COMMON_INCLUDES})
target_compile_options(backbreaker_ppc PRIVATE ${COMMON_FLAGS} -O2 -Wno-everything
    -include ${CMAKE_SOURCE_DIR}/runtime/ppc_hooks.h) # route MMIO stores to the runtime

# Host runtime
file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS
    ${CMAKE_SOURCE_DIR}/runtime/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/kernel/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/gpu/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/gpu/generated/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/apu/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/hid/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/video/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/debug/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/generated/*.cpp)
add_executable(backbreaker ${RUNTIME_SOURCES})
target_include_directories(backbreaker PRIVATE ${CMAKE_SOURCE_DIR}/runtime)

# Build identity (issue #75): which source a binary came from, printed at startup, in the window
# title, in every capture's summary.txt and by --version. cmake/git_version.cmake regenerates the
# header on every build and rewrites it only when the text changed.
find_package(Git QUIET)
set(BUILD_VERSION_HEADER ${CMAKE_BINARY_DIR}/generated/build_version.h)
add_custom_target(build_version ALL
    COMMAND ${CMAKE_COMMAND} -DGIT_EXECUTABLE=${GIT_EXECUTABLE} -DSRC=${CMAKE_SOURCE_DIR}
            -DOUT=${BUILD_VERSION_HEADER} -P ${CMAKE_SOURCE_DIR}/cmake/git_version.cmake
    BYPRODUCTS ${BUILD_VERSION_HEADER}
    COMMENT "build identity (git)")
add_dependencies(backbreaker build_version)
target_include_directories(backbreaker PRIVATE ${CMAKE_BINARY_DIR}/generated)
target_compile_options(backbreaker PRIVATE ${COMMON_FLAGS} -Wall -Wno-unused-function)
# The CPU check runs before anything built for x86-64-v3 can, so it is built for baseline x86-64
# (the later -march wins) and reports a missing instruction set instead of crashing (#134)
set_source_files_properties(${CMAKE_SOURCE_DIR}/runtime/cpu_check.cpp PROPERTIES COMPILE_OPTIONS "-march=x86-64")
find_package(Vulkan REQUIRED)
find_package(SDL3 REQUIRED CONFIG COMPONENTS SDL3)
target_link_libraries(backbreaker PRIVATE backbreaker_ppc Vulkan::Vulkan SDL3::SDL3 shaderc_shared)

# XMA audio decoder (issue #93): FFmpeg's "xmaframes" decoder, built by scripts/build_xma_decoder.sh
# into third_party/ffmpeg-xma (static, decoder-only, LGPL-2.1+). Optional: without it XMA sounds
# play as silence.
set(FFMPEG_XMA ${CMAKE_SOURCE_DIR}/third_party/ffmpeg-xma)
if (EXISTS ${FFMPEG_XMA}/lib/libavcodec.a AND EXISTS ${FFMPEG_XMA}/lib/libavutil.a)
    target_include_directories(backbreaker PRIVATE ${FFMPEG_XMA}/include)
    target_compile_definitions(backbreaker PRIVATE BB_XMA_DECODER=1)
    target_link_libraries(backbreaker PRIVATE ${FFMPEG_XMA}/lib/libavcodec.a ${FFMPEG_XMA}/lib/libavutil.a m)
    message(STATUS "XMA decoder: ${FFMPEG_XMA}")
else()
    message(WARNING "No XMA decoder (third_party/ffmpeg-xma): game sounds will be silent. "
                    "Run scripts/build_xma_decoder.sh, then re-run cmake.")
endif()
# export symbols so crash backtraces show guest function names
target_link_options(backbreaker PRIVATE -rdynamic)

# Shader disassembler / coverage tool for captured Xenos microcode (no game data needed to build)
add_executable(xenos-disasm tools/xenos_disasm.cpp runtime/gpu/disasm.cpp)
target_compile_options(xenos-disasm PRIVATE -Wall)

# Shader translator check tool: Xenos microcode -> GLSL files (compile them with glslc)
add_executable(xenos-translate tools/xenos_translate.cpp runtime/gpu/shader_translator.cpp runtime/gpu/disasm.cpp)
target_compile_options(xenos-translate PRIVATE -Wall)
