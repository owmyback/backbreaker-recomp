cmake_minimum_required(VERSION 3.24) # 3.24: $<LINK_LIBRARY:WHOLE_ARCHIVE,...> for the runtime
project(BackbreakerRecomp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

if (NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Recompiled code requires Clang (configure with -DCMAKE_CXX_COMPILER=clang++)")
endif()

set(XENON ${CMAKE_SOURCE_DIR}/third_party/XenonRecomp)
set(COMMON_INCLUDES ${CMAKE_SOURCE_DIR}/ppc ${XENON}/XenonUtils ${XENON}/thirdparty/simde)
set(COMMON_FLAGS -march=x86-64-v3 -fno-strict-aliasing)
set(PPC_FLAGS ${COMMON_FLAGS} -O2 -Wno-everything)
set(RUNTIME_FLAGS ${COMMON_FLAGS} -Wall -Wno-unused-function)

# The build is split in two (#138): the runtime, a library that needs no game files and can be
# shipped prebuilt as the "runtime kit" (see the end of this file), and the game, which is the
# recompiled ppc/ plus the import glue, both made from the player's own executable, linked with it.
# BB_RUNTIME_ONLY builds the library and the kit alone, on a machine without ppc/.
option(BB_RUNTIME_ONLY "Build only the runtime library and its kit: no ppc/, no game files needed (#138)" OFF)

# Recompiled game code (generated by XenonRecomp into ppc/)
file(GLOB PPC_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/ppc/*.cpp)
if (NOT BB_RUNTIME_ONLY)
    if (NOT PPC_SOURCES)
        message(FATAL_ERROR "ppc/ is empty: scripts/setup.sh recompiles your executable into it first. "
                            "To build only the runtime (no game files needed), configure with -DBB_RUNTIME_ONLY=ON.")
    endif()
    add_library(backbreaker_ppc STATIC ${PPC_SOURCES})
    target_include_directories(backbreaker_ppc PUBLIC ${COMMON_INCLUDES})
    target_compile_options(backbreaker_ppc PRIVATE ${PPC_FLAGS}
        -include ${CMAKE_SOURCE_DIR}/runtime/ppc_hooks.h) # route MMIO stores to the runtime
endif()

# The runtime is compiled against its own copy of the executable's layout (runtime/sdk/ppc_config.h),
# so it doesn't need ppc/. When ppc/ is there, both must say the same: PPC_LOOKUP_FUNC indexes the
# function table with these values, and a runtime built for another layout would call the wrong code.
function(bb_layout header out)
    file(STRINGS ${header} lines REGEX "^#define PPC_(IMAGE|CODE)_(BASE|SIZE)[ \t]")
    set(values)
    foreach (line IN LISTS lines)
        string(REGEX REPLACE "[ \t]+" " " line "${line}")
        list(APPEND values "${line}")
    endforeach()
    list(SORT values)
    set(${out} "${values}" PARENT_SCOPE)
endfunction()
set(RUNTIME_LAYOUT ${CMAKE_SOURCE_DIR}/runtime/sdk/ppc_config.h)
if (NOT BB_RUNTIME_ONLY)
    set(PPC_LAYOUT ${CMAKE_SOURCE_DIR}/ppc/ppc_config.h)
    if (NOT EXISTS ${PPC_LAYOUT})
        message(FATAL_ERROR "ppc/ppc_config.h is missing: re-run the recompile (scripts/setup.sh --force)")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PPC_LAYOUT}) # a new recompile re-checks
    bb_layout(${PPC_LAYOUT} ppc_layout)
    bb_layout(${RUNTIME_LAYOUT} runtime_layout)
    if (NOT ppc_layout STREQUAL runtime_layout)
        message(FATAL_ERROR "ppc/ppc_config.h (${ppc_layout}) doesn't match the layout the runtime is built for, "
                            "runtime/sdk/ppc_config.h (${runtime_layout}): ppc/ comes from another executable.")
    endif()
endif()

# Host runtime. The import glue (runtime/generated/imports.cpp, made from the executable by
# tools/gen_imports.py) belongs to the game's side of the link, so it is not part of the library.
file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS
    ${CMAKE_SOURCE_DIR}/runtime/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/kernel/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/gpu/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/gpu/generated/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/apu/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/hid/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/video/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/debug/*.cpp
    ${CMAKE_SOURCE_DIR}/runtime/generated/*.cpp)
set(IMPORT_GLUE ${CMAKE_SOURCE_DIR}/runtime/generated/imports.cpp)
list(REMOVE_ITEM RUNTIME_SOURCES ${IMPORT_GLUE})
add_library(backbreaker_runtime STATIC ${RUNTIME_SOURCES})
target_include_directories(backbreaker_runtime PRIVATE ${CMAKE_SOURCE_DIR}/runtime
    ${CMAKE_SOURCE_DIR}/runtime/sdk ${XENON}/XenonUtils ${XENON}/thirdparty/simde)

# Build identity (issue #75): which source a binary came from, printed at startup, in the window
# title, in every capture's summary.txt and by --version. cmake/git_version.cmake regenerates the
# header on every build and rewrites it only when the text changed.
find_package(Git QUIET)
set(BUILD_VERSION_HEADER ${CMAKE_BINARY_DIR}/generated/build_version.h)
add_custom_target(build_version ALL
    COMMAND ${CMAKE_COMMAND} -DGIT_EXECUTABLE=${GIT_EXECUTABLE} -DSRC=${CMAKE_SOURCE_DIR}
            -DOUT=${BUILD_VERSION_HEADER} -P ${CMAKE_SOURCE_DIR}/cmake/git_version.cmake
    BYPRODUCTS ${BUILD_VERSION_HEADER}
    COMMENT "build identity (git)")
add_dependencies(backbreaker_runtime build_version)
target_include_directories(backbreaker_runtime PRIVATE ${CMAKE_BINARY_DIR}/generated)
target_compile_options(backbreaker_runtime PRIVATE ${RUNTIME_FLAGS})
# The CPU check runs before anything built for x86-64-v3 can, so it is built for baseline x86-64
# (the later -march wins) and reports a missing instruction set instead of crashing (#134)
set_source_files_properties(${CMAKE_SOURCE_DIR}/runtime/cpu_check.cpp PROPERTIES COMPILE_OPTIONS "-march=x86-64")
find_package(Vulkan REQUIRED)
# Portable build (#134, packaging/portable): SDL3, shaderc and the C++ runtime linked in, so the
# binary needs only glibc, libvulkan.so.1 and a driver from the system. scripts/setup.sh --portable
# builds it in the container; a normal build links the system's shared libraries.
# KIT_LINK is the same list for the runtime kit's game-link.args (one argument per entry, @KIT@
# for the kit's folder): keep the two in step - packaging/installer/link_game.sh is the check.
option(BB_PORTABLE "Link SDL3, shaderc and the C++ runtime statically (scripts/setup.sh --portable)" OFF)
if (BB_PORTABLE)
    find_package(SDL3 REQUIRED CONFIG COMPONENTS SDL3-static)
    target_link_libraries(backbreaker_runtime PUBLIC Vulkan::Vulkan SDL3::SDL3-static shaderc_combined)
    set(GAME_LINK_OPTIONS -static-libstdc++ -static-libgcc)
    find_library(SHADERC_COMBINED shaderc_combined REQUIRED)
    set(KIT_STATIC_LIBS $<TARGET_FILE:SDL3::SDL3-static> ${SHADERC_COMBINED})
    set(KIT_LINK -lvulkan @KIT@/lib/$<TARGET_FILE_NAME:SDL3::SDL3-static> @KIT@/lib/libshaderc_combined.a
        -static-libstdc++ -static-libgcc -pthread)
else()
    find_package(SDL3 REQUIRED CONFIG COMPONENTS SDL3)
    target_link_libraries(backbreaker_runtime PUBLIC Vulkan::Vulkan SDL3::SDL3 shaderc_shared)
    set(GAME_LINK_OPTIONS)
    set(KIT_STATIC_LIBS)
    set(KIT_LINK -lvulkan -lSDL3 -lshaderc_shared)
endif()

# XMA audio decoder (issue #93): FFmpeg's "xmaframes" decoder, built by scripts/build_xma_decoder.sh
# into third_party/ffmpeg-xma (static, decoder-only, LGPL-2.1+). Optional: without it XMA sounds
# play as silence.
set(FFMPEG_XMA ${CMAKE_SOURCE_DIR}/third_party/ffmpeg-xma CACHE PATH "The XMA decoder's install prefix (scripts/build_xma_decoder.sh --prefix)")
if (EXISTS ${FFMPEG_XMA}/lib/libavcodec.a AND EXISTS ${FFMPEG_XMA}/lib/libavutil.a)
    target_include_directories(backbreaker_runtime PRIVATE ${FFMPEG_XMA}/include)
    target_compile_definitions(backbreaker_runtime PRIVATE BB_XMA_DECODER=1)
    target_link_libraries(backbreaker_runtime PUBLIC ${FFMPEG_XMA}/lib/libavcodec.a ${FFMPEG_XMA}/lib/libavutil.a m)
    list(APPEND KIT_STATIC_LIBS ${FFMPEG_XMA}/lib/libavcodec.a ${FFMPEG_XMA}/lib/libavutil.a)
    list(APPEND KIT_LINK @KIT@/lib/libavcodec.a @KIT@/lib/libavutil.a)
    message(STATUS "XMA decoder: ${FFMPEG_XMA}")
else()
    message(WARNING "No XMA decoder (third_party/ffmpeg-xma): game sounds will be silent. "
                    "Run scripts/build_xma_decoder.sh, then re-run cmake.")
endif()
list(APPEND KIT_LINK -lm -rdynamic)

# The game: the import glue, the whole runtime library and ppc/. WHOLE_ARCHIVE is needed because
# the runtime replaces recompiled functions with strong definitions of the same name (play_lock.cpp,
# fixed_seed.cpp, ... override ppc/'s weak sub_ aliases): from an ordinary archive the linker would
# take only the members something asks for, find ppc/'s weak definitions first and silently drop
# the hooks.
if (NOT BB_RUNTIME_ONLY)
    add_executable(backbreaker ${IMPORT_GLUE})
    target_include_directories(backbreaker PRIVATE ${CMAKE_SOURCE_DIR}/runtime)
    target_compile_options(backbreaker PRIVATE ${RUNTIME_FLAGS})
    target_link_libraries(backbreaker PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,backbreaker_runtime>" backbreaker_ppc)
    # export symbols so crash backtraces show guest function names
    target_link_options(backbreaker PRIVATE -rdynamic ${GAME_LINK_OPTIONS})
endif()

# --- The runtime kit (#138) ---------------------------------------------------------------------
# Everything a machine with the player's ppc/ and import glue needs to link the game against this
# runtime without building it:  cmake --install <build> --component runtime --prefix <kit>
#   lib/                              libbackbreaker_runtime.a (+ the static libraries it links)
#   include/                          the headers ppc/ and the import glue are compiled with
#   share/backbreaker-runtime/        version.txt, the runtime's ppc_config.h, and the compile and
#                                     link arguments: one per line, @KIT@ = the kit's folder,
#                                     @PPC@ = the player's ppc/, @GAME@ = the game's own objects
# packaging/installer/link_game.sh uses them. The kit holds no game content: only this repo's
# runtime, its open-source libraries and simde's headers.
set(KIT_SHARE share/backbreaker-runtime)
string(TOUPPER "${CMAKE_BUILD_TYPE}" build_type)
separate_arguments(config_flags UNIX_COMMAND "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${build_type}}")
if (DEFINED CMAKE_CXX_EXTENSIONS AND NOT CMAKE_CXX_EXTENSIONS)
    set(std_flag ${CMAKE_CXX20_STANDARD_COMPILE_OPTION})
else()
    set(std_flag ${CMAKE_CXX20_EXTENSION_COMPILE_OPTION})
endif()
set(KIT_PPC_COMPILE ${config_flags} ${std_flag} ${PPC_FLAGS} -include @KIT@/include/ppc_hooks.h
    -I@PPC@ -I@KIT@/include/simde)
set(KIT_IMPORTS_COMPILE ${config_flags} ${std_flag} ${RUNTIME_FLAGS} -I@KIT@/include -I@PPC@
    -I@KIT@/include/simde)
set(KIT_GAME_LINK -Wl,--whole-archive @KIT@/lib/libbackbreaker_runtime.a -Wl,--no-whole-archive
    @GAME@ ${KIT_LINK})
foreach (step ppc-compile imports-compile game-link)
    string(TOUPPER "KIT_${step}" var)
    string(REPLACE "-" "_" var "${var}")
    list(JOIN ${var} "\n" text)
    file(GENERATE OUTPUT ${CMAKE_BINARY_DIR}/kit-args/${step}.args CONTENT "${text}\n")
    install(FILES ${CMAKE_BINARY_DIR}/kit-args/${step}.args DESTINATION ${KIT_SHARE} COMPONENT runtime)
endforeach()
install(TARGETS backbreaker_runtime ARCHIVE DESTINATION lib COMPONENT runtime)
if (KIT_STATIC_LIBS)
    install(FILES ${KIT_STATIC_LIBS} DESTINATION lib COMPONENT runtime)
endif()
install(FILES runtime/ppc_hooks.h runtime/memory.h DESTINATION include COMPONENT runtime)
install(FILES runtime/kernel/import_support.h DESTINATION include/kernel COMPONENT runtime)
install(DIRECTORY ${XENON}/thirdparty/simde/ DESTINATION include/simde COMPONENT runtime
    FILES_MATCHING PATTERN "*.h" PATTERN "COPYING")
install(FILES ${RUNTIME_LAYOUT} DESTINATION ${KIT_SHARE} COMPONENT runtime)
# version.txt: the same "<revision> (<branch>, built <date>)" line the game prints with --version
install(CODE "set(BB_VERSION_HEADER \"${BUILD_VERSION_HEADER}\")\nset(BB_KIT_SHARE \"${KIT_SHARE}\")" COMPONENT runtime)
install(SCRIPT cmake/kit_version.cmake COMPONENT runtime)

# The installer's game-file steps (#138, installer/): checks the player's image and does what
# tools/xdvdfs.py, xex_decrypt.py, gen_imports.py and title_icon.py do, without Python or OpenSSL.

set(INSTALLER_SOURCES installer/crypto.cpp installer/disc.cpp installer/xex.cpp installer/game_check.cpp)
add_executable(backbreaker-installer installer/main.cpp ${INSTALLER_SOURCES})
target_include_directories(backbreaker-installer PRIVATE ${XENON}/XenonUtils)
target_compile_options(backbreaker-installer PRIVATE -Wall -Wextra)
